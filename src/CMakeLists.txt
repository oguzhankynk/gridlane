configure_file(version.cpp.in "${CMAKE_CURRENT_BINARY_DIR}/version.cpp" @ONLY)

add_library(gridlane
  "${CMAKE_CURRENT_BINARY_DIR}/version.cpp"
  runtime/layout.cpp
  runtime/fabric.cpp
  runtime/collective.cpp
)
add_library(gridlane::gridlane ALIAS gridlane)

target_compile_features(gridlane PUBLIC cxx_std_17)

target_include_directories(gridlane
  PUBLIC
    $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)

target_link_libraries(gridlane PUBLIC pushseg::pushseg pairring::pairring wireforge::runtime)

gridlane_set_warnings(gridlane)
gridlane_enable_sanitizers(gridlane)
gridlane_enable_coverage(gridlane)
gridlane_enable_clang_tidy(gridlane)
gridlane_enable_cppcheck(gridlane)

# Control plane codecs generated by wireforge from model/project.yaml, one
# library per unit. The generated code is committed; see model/README.md.
foreach(unit IN ITEMS coordinator worker)
  add_library(gridlane_control_${unit} INTERFACE)
  add_library(gridlane::control_${unit} ALIAS gridlane_control_${unit})
  target_include_directories(gridlane_control_${unit} INTERFACE
    $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/generated/nodes/${unit}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_link_libraries(gridlane_control_${unit} INTERFACE wireforge::runtime)
  install(TARGETS gridlane_control_${unit} EXPORT gridlaneTargets)
  install(DIRECTORY "${PROJECT_SOURCE_DIR}/generated/nodes/${unit}/gridlane_control_${unit}"
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
endforeach()
